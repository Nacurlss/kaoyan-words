- [ ] **Step 5: 重写 `_get_exclude_words` 为差集逻辑**

将 `_get_exclude_words` 函数（第 47-60 行）改为：

```python
def _get_exclude_words(levels: list[str]) -> set[str]:
    """Get exclude word set using difference logic.
    Each level only excludes words unique to that level
    (i.e., not already covered by lower levels):
    - primary: primary school words
    - zhongkao: all zhongkao words
    - gaokao: gaokao words NOT in zhongkao/primary
    - cet4: cet4 words NOT in gaokao/zhongkao/primary
    - custom: custom words
    """
    # Load all word lists lazily
    all_primary = set()
    all_zhongkao = set()
    all_gaokao = set()
    all_cet4 = set()
    all_custom = set()

    def _load(name: str) -> set[str]:
        path = WORDS_DIR / name
        if path.exists():
            return load_word_list(str(path))
        return set()

    # Preload as needed
    needed = set(levels)
    if "primary" in needed:
        all_primary = _load("小学英语大纲词汇.txt")
    if "zhongkao" in needed or "gaokao" in needed or "cet4" in needed:
        all_zhongkao = _load("zhongkao.txt")
    if "gaokao" in needed or "cet4" in needed:
        all_gaokao = _load("gaokao.txt")
    if "cet4" in needed:
        all_cet4 = _load("cet4.txt")
    if "custom" in needed:
        all_custom = _load("custom.txt")

    exclude = set()
    if "primary" in levels:
        exclude.update(all_primary)
    if "zhongkao" in levels:
        exclude.update(all_zhongkao)
    if "gaokao" in levels:
        gaokao_only = all_gaokao - all_zhongkao - all_primary
        exclude.update(gaokao_only)
    if "cet4" in levels:
        cet4_only = all_cet4 - all_gaokao - all_zhongkao - all_primary
        exclude.update(cet4_only)
    if "custom" in levels:
        exclude.update(all_custom)

    return exclude
```

用以下内容替换 `backend/dictionary.py` 全部：

```python
import json
from pathlib import Path

import httpx

CACHE_PATH = Path(__file__).parent / "wordlists" / "dict_cache.json"
API_URL = "https://api.dictionaryapi.dev/api/v2/entries/en/"

_cache = None


def _load_cache() -> dict:
    global _cache
    if _cache is None:
        if CACHE_PATH.exists():
            try:
                _cache = json.loads(CACHE_PATH.read_text(encoding="utf-8"))
            except (json.JSONDecodeError, OSError):
                _cache = {}
        else:
            _cache = {}
    return _cache


def _save_cache(cache: dict):
    try:
        CACHE_PATH.write_text(
            json.dumps(cache, ensure_ascii=False, indent=2), encoding="utf-8"
        )
    except OSError:
        pass


async def lookup_word(word: str) -> list[dict]:
    """Look up a word. Returns list of {pos, meaning} dicts.
    Checks local cache first, then queries Free Dictionary API."""
    cache = _load_cache()
    key = word.lower()
    if key in cache:
        return cache[key]

    try:
        async with httpx.AsyncClient(timeout=5.0) as client:
            resp = await client.get(f"{API_URL}{key}")
            if resp.status_code == 200:
                data = resp.json()
                meanings = data[0].get("meanings", []) if data else []
                result = []
                for m in meanings:
                    pos = m.get("partOfSpeech", "")
                    for d in m.get("definitions", [])[:2]:
                        result.append({
                            "pos": pos,
                            "meaning": d.get("definition", ""),
                        })
                cache[key] = result
                _save_cache(cache)
                return result
    except Exception:
        pass

    return []


def annotate_word_index(word_index: dict) -> dict:
    """Add dictionary meanings to word index entries (synchronous stub).
    For async lookups, use lookup_word() directly in endpoints."""
    return word_index
```

async def get_word_detail(lemma: str):
    """Get detailed info for a single word."""
    global _session_word_index
    info = _session_word_index.get(lemma)
    if not info:
        # Try dictionary lookup even if not in index
        meanings = await lookup_word(lemma)
        return {
            "lemma": lemma,
            "variants": [],
            "count": 0,
            "paper_count": 0,
            "frequency": 0,
            "papers": [],
            "meanings": meanings,
            "sentences": {},
        }

    # Add dictionary meanings if missing
    if not info.get("meanings"):

    return {

        "variants": info["variants"],
        "count": info["count"],
        "paper_count": info.get("paper_count", len(info.get("paper_set", []))),
        "frequency": round(info["frequency"], 4),

        "meanings": info.get("meanings", []),

- [ ] **Step 14: 重写 SettingsPanel.tsx——基础词开关+可折叠高级设置**

用以下内容替换 `frontend/src/components/SettingsPanel.tsx` 全部：

```tsx
import { useState, useEffect } from "react";
import { updateSettings } from "../api";
import type { Settings } from "../types";

interface Props {
  settings: Settings;
  onChange: (s: Settings) => void;
}

const ADVANCED_LEVELS = [
  { key: "primary", label: "小学词汇" },
  { key: "zhongkao", label: "中考词汇" },
  { key: "gaokao", label: "高考词汇" },
  { key: "cet4", label: "四级词汇" },
  { key: "custom", label: "自定义词表" },
];

export default function SettingsPanel({ settings, onChange }: Props) {
  const [highThreshold, setHighThreshold] = useState(settings.high_threshold);
  const [mediumThreshold, setMediumThreshold] = useState(settings.medium_threshold);
  const [excludeLevels, setExcludeLevels] = useState(settings.exclude_levels);
  const [showAdvanced, setShowAdvanced] = useState(false);

  // Sync from props when settings change externally
  useEffect(() => {
    setHighThreshold(settings.high_threshold);
    setMediumThreshold(settings.medium_threshold);
    setExcludeLevels(settings.exclude_levels);
  }, [settings]);

  const basicEnabled = excludeLevels.includes("primary") && excludeLevels.includes("zhongkao");

  const applyExcludeLevels = async (next: string[]) => {
    setExcludeLevels(next);
    const newSettings = { ...settings, high_threshold: highThreshold, medium_threshold: mediumThreshold, exclude_levels: next };
    await updateSettings(newSettings);
    onChange(newSettings);
  };

  const toggleBasic = async () => {
    if (basicEnabled) {
      applyExcludeLevels(excludeLevels.filter((l: string) => l !== "primary" && l !== "zhongkao"));
    } else {
      const next = [...new Set([...excludeLevels, "primary", "zhongkao"])];
      applyExcludeLevels(next);
    }
  };

  const toggleLevel = async (key: string) => {
    const next = excludeLevels.includes(key)
      ? excludeLevels.filter((l: string) => l !== key)
      : [...excludeLevels, key];
    applyExcludeLevels(next);
  };

  const applyThresholds = async () => {
    const newSettings = { ...settings, high_threshold: highThreshold, medium_threshold: mediumThreshold, exclude_levels: excludeLevels };
    await updateSettings(newSettings);
    onChange(newSettings);
  };

  return (
    <div className="settings-panel">
      <h3>频率设置</h3>
      <label>
        高频线
        <span className="threshold-val">{highThreshold.toFixed(2)}</span>
      </label>
      <input
        type="range" min="0" max="2" step="0.05" value={highThreshold}
        onChange={(e) => setHighThreshold(parseFloat(e.target.value))}
        onMouseUp={applyThresholds}
      />

      <label>
        中频线
        <span className="threshold-val">{mediumThreshold.toFixed(2)}</span>
      </label>
      <input
        type="range" min="0" max="2" step="0.05" value={mediumThreshold}
        onChange={(e) => setMediumThreshold(parseFloat(e.target.value))}
        onMouseUp={applyThresholds}
      />

      <h3 style={{ marginTop: 16 }}>基础词过滤</h3>
      <label className="basic-toggle">
        <input type="checkbox" checked={basicEnabled} onChange={toggleBasic} />
        筛去基础词（小学+中考）
      </label>

      <div className="advanced-toggle" onClick={() => setShowAdvanced(!showAdvanced)}>
        {showAdvanced ? "▼" : "▶"} 高级设置
      </div>
      {showAdvanced && (
        <div className="level-checkboxes">
          {ADVANCED_LEVELS.map(({ key, label }) => (
            <label key={key}>
              <input
                type="checkbox"
                checked={excludeLevels.includes(key)}
                onChange={() => toggleLevel(key)}
              />
              {label}
            </label>
          ))}
        </div>
      )}
    </div>
  );
}
```
